domains = ["example.com", "www.example.com", "api.example.com", "cdn.example.com"]

1.2 ПОРТ-СКАН (TCP CONNECT)
────────────────────────────

import socket

target_ip = socket.gethostbyname("example.com")
ports = [80, 443, 8080, 8443, 22, 1935, 554, 9000, 3000, 5000, 7000, 8000]

for port in ports:
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        s.settimeout(2)
        result = s.connect_ex((target_ip, port))
        status = "OPEN" if result == 0 else "closed"
        print(f"  {port}/tcp -> {status}")
        s.close()
    except Exception as e:
        print(f"  {port}/tcp -> ERROR: {e}")

5.3 ДЕКОДИРОВАНИЕ ЗАШИФРОВАННЫХ ССЫЛОК
───────────────────────────────────────

# ПРИМЕР 1: Kodik — Caesar(-8) + base64
def kodik_decode(encoded):
    def caesar_shift(text, shift=-8):
        result = []
        for char in text:
            if char.isalpha():
                base = 65 if char.isupper() else 97
                result.append(chr((ord(char) - base + shift) % 26 + base))
            else:
                result.append(char)
        return ''.join(result)

    caesar = caesar_shift(encoded, -8)
    padded = caesar + '=' * (4 - len(caesar) % 4)
    decoded = base64.b64decode(padded).decode('utf-8')
    return 'https:' + decoded if decoded.startswith('//') else decoded

# ПРИМЕР 2: Base64 без padding
def base64_nopadding_decode(encoded):
    padded = encoded + '=' * (4 - len(encoded) % 4)
    return base64.b64decode(padded).decode('utf-8')

# ПРИМЕР 3: Reverse + base64
def reverse_base64_decode(encoded):
    reversed_str = encoded[::-1]
    padded = reversed_str + '=' * (4 - len(reversed_str) % 4)
    return base64.b64decode(padded).decode('utf-8')

# ПРИМЕР 4: XOR brute-force
def xor_bruteforce(data, keys=[b'key', b'secret', b'video']):
    for key in keys:
        xored = bytes([b ^ key[i % len(key)] for i, b in enumerate(data)])
        text = xored.decode('utf-8', errors='ignore')
        if 'http' in text.lower():
            return text
    return None

# ПРИМЕР 5: Brute-force Caesar + base64
def caesar_bruteforce(encoded):
    for shift in range(-25, 26):
        if shift == 0: continue
        shifted = ''.join(
            chr((ord(c) - 65 + shift) % 26 + 65) if c.isupper() else
            chr((ord(c) - 97 + shift) % 26 + 97) if c.islower() else c
            for c in encoded
        )
        try:
            padded = shifted + '=' * (4 - len(shifted) % 4)
            decoded = base64.b64decode(padded).decode('utf-8')
            if any(k in decoded.lower() for k in ['http', '//', '.mp4', '.m3u8']):
                return shift, decoded
        except:
            pass
    return None, None
